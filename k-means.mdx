# k-means

Given $K$ centroids:
1. **Initialize centroids**, preferably with K-means++ rather than purely random points.
2. **Assign each point to its closest centroid:**  
   
$$
z_i = \arg\min_k \left\|x_i - \mu_k\right\|_2^2
$$

3. **Update each centroid using the mean of the points assigned to it:**

$$
\mu_k = \frac{1}{n_k}\sum_{i:z_i=k} x_i
$$

4. **Repeat** until the assignments or centroids stop changing, or a maximum iteration count is reached.

Each assignment/update cycle does not increase the K-means objective, so the algorithm converges to a **local optimum**, not necessarily the global optimum.


```python
class KMeansClustering:
    def __init__(self, k=3):
        self.k = k
        self.centroids = None
    
    
    def eculiden_dis(data_point, centroids):
        return np.sqrt(np.sum((centroids-data_point)**2, axis=1))
    
    def fit(self, X, max_iter=200):
        self.centroids = np.random.uniform(np.amin(X,axis=0), np.amax(X,axis=0),
                                          size=(self.k,X.shape[1]))

        #iterate and reach threshold
        for _ in range(max_iter):
            y = [] # cluster labels
            
            #assign the data point to cluster_num
            for data_point in X:
                dist = KMeansClustering.eculiden_dis(data_point, self.centroids)
                cluster_num = np.argmin(dist)
                y.append(cluster_num)
            y=np.array(y)
            
            
            cluster_indices = []            
            for i in range(self.k):
                cluster_indices.append(np.argwhere(y==i))
                
            
            cluster_centers = []            
            for i, indices in enumerate(cluster_indices):
                if len(indices) == 0:
                    cluster_centers.append(self.centroids[i]) 
                else:
                    cluster_centers.append(np.mean(X[indices], axis=0)[0])
                    
            if np.max(self.centroids-np.array(cluster_centers))<=0.0001:
                break
            else:
                self.centroids = np.array(cluster_centers)
            
        return y
    
    
random_points = np.random.randint(0,100,(100,2))
kmeans = KMeansClustering(k=10)
labels = kmeans.fit(random_points)

plt.scatter(random_points[:,0], random_points[:,1],c=labels)
plt.scatter(kmeans.centroids[:,0],kmeans.centroids[:,1],c=range(len(kmeans.centroids)),
           marker="*",s=200)
plt.show
```
